"""MyŠkoda MyŠkoda authentication (MyŠkoda sign-in secure login → mysmob CONNECT token)."""

from __future__ import annotations

import base64
import hashlib
import re
import secrets
import uuid
from urllib.parse import urlencode, urljoin, urlparse, parse_qs

import httpx

MYSMOB_BASE = "https://MyŠkoda API"
IDENTITY = "https://<identity_host>"
TOKEN_TYPE = "CONNECT"
CLIENT_ID = "<client_id>"
X_CLIENT_ID = "<x_client_id>"
SCOPE = (
    "address badge birthdate cars driversLicense dealers email mileage mbb "
    "nationalIdentifier openid phone profession profile vin"
)
REDIRECT_URI = "myskoda://redirect/login/"
UA = "<user_agent>"


class MyŠkodaAuthError(Exception):
    """Base auth failure."""


class MyŠkodaAuthInvalidCredentials(MyŠkodaAuthError):
    """Wrong password / account rejected."""


class MyŠkodaAuthExpired(MyŠkodaAuthError):
    """Refresh token invalid or expired."""


def _pkce() -> tuple[str, str]:
    chars = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"
    verifier = "".join(secrets.choice(chars) for _ in range(64))
    raw = base64.b64encode(verifier.encode()).decode()
    verifier = raw.replace("+", "-").replace("/", "_").replace("=", "")
    challenge = base64.urlsafe_b64encode(hashlib.sha256(verifier.encode()).digest()).decode().rstrip("=")
    return verifier, challenge


def _nonce() -> str:
    digest = hashlib.sha256(str(uuid.uuid4()).encode()).digest()
    return base64.b64encode(digest).decode().rstrip("=")


def _hidden(html: str) -> dict[str, str]:
    out: dict[str, str] = {}
    for m in re.finditer(
        r"<input (?=[^>]* name=[\"']([^\"']*)|)(?=[^>]* value=[\"']([^\"']*)|)", html
    ):
        if m.group(1):
            out[m.group(1)] = m.group(2) or ""
    return out


def bearer_headers(access_token: str) -> dict[str, str]:
    return {
        "accept": "application/json",
        "user-agent": UA,
        "accept-language": "en",
        "authorization": f"Bearer {access_token}",
        "x-demo-mode": "false",
    }


async def exchange_tokens(*, code: str, verifier: str) -> dict:
    async with httpx.AsyncClient(timeout=60.0) as client:
        resp = await client.post(
            f"{MYSMOB_BASE}/api/v1/authentication/exchange-authorization-code",
            params={"tokenType": TOKEN_TYPE},
            json={"code": code, "redirectUri": REDIRECT_URI, "verifier": verifier},
            headers={"accept": "*/*", "content-type": "application/json", "user-agent": UA},
        )
    if resp.status_code >= 400:
        raise MyŠkodaAuthError(f"token exchange failed: HTTP {resp.status_code}")
    data = resp.json()
    if not data.get("accessToken"):
        raise MyŠkodaAuthError("token exchange returned no accessToken")
    return data


async def refresh_tokens(refresh_token: str) -> dict:
    async with httpx.AsyncClient(timeout=60.0) as client:
        resp = await client.post(
            f"{MYSMOB_BASE}/api/v1/authentication/refresh-token",
            params={"tokenType": TOKEN_TYPE},
            json={"token": refresh_token},
            headers={"accept": "*/*", "content-type": "application/json", "user-agent": UA},
        )
    if resp.status_code >= 400:
        raise MyŠkodaAuthExpired(f"refresh failed: HTTP {resp.status_code}")
    data = resp.json()
    if not data.get("accessToken"):
        raise MyŠkodaAuthExpired("refresh returned no accessToken")
    return data


async def login_with_password(email: str, password: str) -> dict:
    """Full MyŠkoda login. Raises MyŠkodaAuthInvalidCredentials on bad password."""
    verifier, challenge = _pkce()
    params = {
        "client_id": CLIENT_ID,
        "scope": SCOPE,
        "response_type": "code",
        "redirect_uri": REDIRECT_URI,
        "nonce": _nonce(),
        "state": str(uuid.uuid4()),
        "code_challenge": challenge,
        "code_challenge_method": "S256",
    }
    headers = {
        "User-Agent": UA,
        "Accept": "text/html,application/xhtml+xml",
        "x-requested-with": "cz.skodaauto.connect",
        "x-app-name": "cz.skodaauto.connect",
        "x-app-version": "8.0.0",
        "x-brand": "skoda",
        "x-client-id": X_CLIENT_ID,
    }
    async with httpx.AsyncClient(follow_redirects=False, timeout=60.0, headers=headers) as client:
        resp = await client.get(f"{IDENTITY}/oidc/v1/authorize", params=params)
        loc = resp.headers.get("location")
        if not loc:
            raise MyŠkodaAuthError("authorize did not redirect")
        cur = loc if loc.startswith("http") else urljoin(str(resp.url), loc)
        body = ""
        for _ in range(12):
            r = await client.get(cur)
            if r.status_code in (301, 302, 303, 307, 308) and r.headers.get("location"):
                nxt = r.headers["location"]
                cur = nxt if nxt.startswith("http") else urljoin(str(r.url), nxt)
                continue
            body = r.text
            break
        if "emailPasswordForm" not in body:
            raise MyŠkodaAuthError("login form not found")
        form = _hidden(body)
        form["email"] = email
        action_m = re.search(r'<form[^>]+action=["\']([^"\']+)', body)
        ident_url = urljoin(cur, action_m.group(1)) if action_m else f"{IDENTITY}/signin-service/v1/{CLIENT_ID}/login/identifier"
        r = await client.post(ident_url, data=form, headers={"Content-Type": "application/x-www-form-urlencoded"})
        body2 = r.text
        hidden = _hidden(body2)
        csrf = re.search(r"csrf_token: '([^']+)'", body2)
        hmac_m = re.search(r'"hmac":"([^"]+)"', body2)
        relay_m = re.search(r'"relayState":"([^"]+)"', body2)
        form2 = {
            "_csrf": (csrf.group(1) if csrf else hidden.get("_csrf", "")),
            "email": email,
            "password": password,
            "hmac": (hmac_m.group(1) if hmac_m else hidden.get("hmac", "")),
            "relayState": (relay_m.group(1) if relay_m else hidden.get("relayState", "")),
        }
        if not form2["_csrf"] and not form2["hmac"]:
            raise MyŠkodaAuthError("could not parse password form")
        action2 = re.search(r'<form[^>]+action=["\']([^"\']+)', body2)
        auth_url = urljoin(cur, action2.group(1)) if action2 else f"{IDENTITY}/signin-service/v1/{CLIENT_ID}/login/authenticate"
        r2 = await client.post(auth_url, data=form2, headers={"Content-Type": "application/x-www-form-urlencoded"})
        loc2 = r2.headers.get("location")
        if r2.status_code == 401 or (loc2 and "error" in loc2 and "code=" not in loc2):
            raise MyŠkodaAuthInvalidCredentials("MyŠkoda rejected email or password")
        if not loc2:
            if r2.status_code >= 400:
                raise MyŠkodaAuthInvalidCredentials("MyŠkoda rejected email or password")
            raise MyŠkodaAuthError("authenticate did not redirect")
        next_url = loc2 if loc2.startswith("http") else urljoin(str(r2.url), loc2)
        code = None
        for _ in range(12):
            if "code=" in next_url:
                code = (parse_qs(urlparse(next_url).query).get("code") or [None])[0]
                if code:
                    break
            if next_url.startswith("myskoda:"):
                code = (parse_qs(urlparse(next_url).query).get("code") or [None])[0]
                if code:
                    break
            rr = await client.get(next_url)
            if rr.status_code in (301, 302, 303, 307, 308) and rr.headers.get("location"):
                nxt = rr.headers["location"]
                next_url = nxt if nxt.startswith("http") else urljoin(str(rr.url), nxt)
                continue
            m = re.search(r"[?&#]code=([A-Za-z0-9._%-]+)", rr.text or "")
            code = m.group(1) if m else None
            break
        if not code:
            raise MyŠkodaAuthError("authorization code not captured")
    return await exchange_tokens(code=code, verifier=verifier)
```

class SkodaMyŠkodaError(Exception):
    def __init__(self, message: str, status_code: int | None = None) -> None:
        super().__init__(message)
        self.status_code = status_code


class SkodaMyŠkodaClient:
    def __init__(self, access_token: str) -> None:
        self._token = access_token
        self._client = httpx.AsyncClient(
            base_url=MYSMOB_BASE,
            timeout=30.0,
            headers=bearer_headers(access_token),
        )

    async def _get(self, path: str) -> dict:
        resp = await self._client.get(path)
        if resp.status_code >= 400:
            raise SkodaMyŠkodaError(f"MyŠkoda GET {path} -> HTTP {resp.status_code}", resp.status_code)
        return resp.json()

    async def get_charging(self, vin: str) -> dict:
        return await self._get(f"/api/v1/charging/{vin}")

    async def get_vehicle_status(self, vin: str) -> dict:
        return await self._get(f"/api/v2/vehicle-status/{vin}")

    async def get_driving_range(self, vin: str) -> dict:
        return await self._get(f"/api/v2/vehicle-status/{vin}/driving-range")

    async def get_parking(self, vin: str) -> dict:

    async def get_readiness(self, vin: str) -> dict:

SourceKind = Literal["legacy", "public", "login"]  # replace existing Literal


def snapshot_from_bff(
    *,
    charging: dict,
    vehicle_status: dict,
    parking: dict,
    driving_range: dict | None = None,
    readiness: dict | None = None,
    now: datetime,
    vin: str | None = None,
) -> NormalizedSnapshot:
    """Map mysmob MyŠkoda payloads → NormalizedSnapshot (same contract as public)."""
    snap = NormalizedSnapshot(captured_at=now, source="login", vin=vin)
    st = (charging or {}).get("status") or {}
    batt = st.get("battery") or {}
    if batt.get("stateOfChargeInPercent") is not None:
        snap.soc = float(batt["stateOfChargeInPercent"])
    if batt.get("remainingCruisingRangeInMeters") is not None:
        snap.remaining_range_m = float(batt["remainingCruisingRangeInMeters"])
    if st.get("chargePowerInKw") is not None:
        snap.charge_power_kw = float(st["chargePowerInKw"])
    snap.charge_state = st.get("state")
    snap.charge_type = st.get("chargeType")
    settings = (charging or {}).get("settings") or {}
    if settings.get("targetStateOfChargeInPercent") is not None:
        snap.target_soc = float(settings["targetStateOfChargeInPercent"])

    if snap.soc is None and driving_range:
        eng = driving_range.get("primaryEngineRange") or {}
        if eng.get("currentSoCInPercent") is not None:
            snap.soc = float(eng["currentSoCInPercent"])
        if eng.get("remainingRangeInKm") is not None:
            snap.range_km = float(eng["remainingRangeInKm"])

    park = (parking or {}).get("parkingPosition") or {}
    gps = park.get("gpsCoordinates") or {}
    if gps.get("latitude") is not None:
        snap.lat = float(gps["latitude"])
        snap.lon = float(gps.get("longitude") or 0)
    if readiness is not None:
        snap.is_online = not bool(readiness.get("unreachable"))
        snap.is_moving = bool(readiness.get("inMotion"))
        snap.is_parked = not snap.is_moving
    else:
        snap.is_online = True

    vs = (vehicle_status or {}).get("overall") or {}
    snap.extra["doors_locked"] = vs.get("doorsLocked") or vs.get("locked")
    snap.extra["doors"] = vs.get("doors")
    snap.extra["windows"] = vs.get("windows")
    return snap
```
